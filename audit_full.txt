def comparar(texto, nombre_corto=True):
    '''Muestra c├│mo cada tokenizador fragmenta el mismo texto.'''
    filas = []
    for model_id, tok in tokenizadores.items():
        nombre = model_id.split("/")[-1] if nombre_corto else model_id
        piezas = tok.tokenize(texto)
        filas.append({
            "Modelo":   nombre,
            "N tokens": len(piezas),
            "Tokens":   " | ".join(piezas),
        })
    return pd.DataFrame(filas).set_index("Modelo")

Se usa el **Amazon Shopping Queries Dataset (ESCI)**, publicado por Amazon Science bajo

| Atributo | Detalle |
|---|---|
| **Fuente** | [amazon-science/esci-data](https://github.com/amazon-science/esci-data) |
| **Licencia** | Apache 2.0 |
| **Idiomas** | Ingl├⌐s, Espa├▒ol, Japon├⌐s |

| **Etiquetas** | `E` Exact ┬╖ `S` Substitute ┬╖ `C` Complement ┬╖ `I` Irrelevant |
| **Campos usados** | `query`, `product_title`, `product_locale`, `esci_label`, `split` |
| **Splits** | `train` y `test` predefinidos en el dataset |

**Mapeo de etiquetas ESCI a formato binario para el encoder de clasificaci├│n:**

| ESCI | Significado | Etiqueta modelo |
|---|---|---|
| `E` (Exact) | Producto que responde directamente la consulta | `relevante` (1) |
| `S` (Substitute) | Producto alternativo, posiblemente ├║til | `no relevante` (0) |
| `C` (Complement) | Producto complementario | `no relevante` (0) |
| `I` (Irrelevant) | Sin relaci├│n con la consulta | `no relevante` (0) |

Se usa el mapeo binario `E ΓåÆ relevante`, `S+C+I ΓåÆ no relevante` para simplificar la tarea
en M1. La p├⌐rdida de informaci├│n en `S` y `C` se documenta como limitaci├│n conocida (┬º2.8).

    '''Normalizaci├│n reproducible: min├║sculas, colapso de espacios.'''
    if not isinstance(texto, str):
        return ""
    texto = texto.lower()
    texto = re.sub(r"\s+", " ", texto).strip()
    return texto

def n_palabras(texto: str) -> int:
    '''Cuenta palabras en un texto ya limpio.'''
    return len(texto.split()) if isinstance(texto, str) else 0

## 2.8 ┬╖ Limitaciones conocidas del dataset

Las siguientes limitaciones se documentan expl├¡citamente antes de usar el dataset en
el entrenamiento. Ignorarlas ser├¡a un error de dise├▒o que afectar├¡a la interpretaci├│n
de los resultados.

**1. Origen del dataset ΓÇö Amazon US/ES**
El dataset proviene de b├║squedas reales en Amazon, que opera principalmente en mercados
de Espa├▒a y Estados Unidos hispanohablantes. El cat├ílogo de Digitdeck puede tener
distribuciones diferentes (categor├¡as distintas, marcas propias, terminolog├¡a local
colombiana). El modelo puede rendir peor en categor├¡as subrepresentadas en el dataset ESCI.

**2. Mapeo binario pierde informaci├│n**
El mapeo `E ΓåÆ relevante`, `S+C+I ΓåÆ no relevante` trata igual un producto sustituto (├║til)
y uno completamente irrelevante. Esto puede hacer que el modelo aprenda se├▒ales m├ís ruidosas
para casos ambiguos (`S` y `C`). La versi├│n de 4 clases queda como mejora futura.

**3. Desbalance de clases**
El dataset no est├í balanceado: hay m├ís pares no relevantes que relevantes. El entrenamiento
sin correcci├│n tender├í a predecir `no relevante` m├ís frecuentemente. Se mitiga en ┬º3 con
`pos_weight` / pesos de clase en la funci├│n de p├⌐rdida.

**4. Texto de producto truncado**
Solo se usa `product_title`. La descripci├│n completa (`product_description`) contiene
informaci├│n relevante pero encarece el entrenamiento. Para M1 se prioriza la eficiencia
sobre la completitud.

def promedio_tokens(salida, attention_mask):
    '''Mean pooling sobre tokens no-padding ΓÇö m├⌐todo recomendado en la model card de E5.'''
    embeddings = salida.last_hidden_state
    mask = attention_mask.unsqueeze(-1).expand(embeddings.size()).float()
    return (embeddings * mask).sum(1) / mask.sum(1).clamp(min=1e-9)

@torch.no_grad()
def embeber(textos, batch_size=32):
    '''Codifica una lista de textos (ya con prefijo query:/passage:) a embeddings L2-normalizados.'''
    salidas = []
    for i in range(0, len(textos), batch_size):
        lote = textos[i:i + batch_size]
        entradas = tok_base(lote, padding=True, truncation=True, max_length=64, return_tensors="pt").to(DEVICE)

        emb = promedio_tokens(salida, entradas["attention_mask"])
        emb = F.normalize(emb, p=2, dim=1)
        salidas.append(emb.cpu())
    return torch.cat(salidas, dim=0)

def ndcg_at_k(relevancias_ordenadas, k=10):
    '''nDCG@k para relevancia binaria, dado un ranking ya ordenado por score descendente.'''
    relevancias_ordenadas = np.asarray(relevancias_ordenadas[:k], dtype=float)
    descuentos = 1.0 / np.log2(np.arange(2, len(relevancias_ordenadas) + 2))
    dcg = float(np.sum(relevancias_ordenadas * descuentos))
    ideal = np.sort(relevancias_ordenadas)[::-1]
    idcg = float(np.sum(ideal * descuentos))
    return dcg / idcg if idcg > 0 else 0.0

def mrr(relevancias_ordenadas):
    '''Reciprocal rank de la primera posici├│n relevante.'''
    for i, rel in enumerate(relevancias_ordenadas):
        if rel > 0:
            return 1.0 / (i + 1)
    return 0.0

def recall_at_k(relevancias_ordenadas, total_relevantes, k=10):

def medir_guardrails_encoder(fn_scorear, df_muestra, n_reps=50):
    '''Mide pico de VRAM y p95 de latencia de inferencia sobre un lote peque├▒o representativo.'''
    if DEVICE != "cuda":
        return {"memoria_pico_mb": None, "p95_latencia_ms": None, "nota": "sin GPU ΓÇö no medible"}

    torch.cuda.reset_peak_memory_stats()
    torch.cuda.synchronize()

    tiempos = []
    for _ in range(n_reps):
        t0 = time.perf_counter()
        fn_scorear(df_muestra)
        torch.cuda.synchronize()
        tiempos.append((time.perf_counter() - t0) * 1000)  # ms

    memoria_pico_mb = torch.cuda.max_memory_allocated() / (1024 ** 2)
    p95 = float(np.percentile(tiempos, 95))
    return {"memoria_pico_mb": round(memoria_pico_mb, 1), "p95_latencia_ms": round(p95, 2)}
